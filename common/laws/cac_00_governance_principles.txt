# group = this is the law_group a law belongs to
# icon = graphical icon shown in-game
# modifier = {} modifier on country for having adopted this law

law_chiefdom = {
	group = lawgroup_governance_principles

	icon = "gfx/interface/icons/law_icons/chiefdom.dds"

	progressiveness = -100

	can_enact = {
		is_country_type = decentralized
	}

	is_visible = {
		is_country_type = decentralized
	}

	modifier = {
		country_legitimacy_headofstate_add = 20
		country_authority_add = 100
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

law_monarchy = {
	group = lawgroup_governance_principles

	icon = "gfx/interface/icons/law_icons/monarchy.dds"

	progressiveness = 0

	can_enact = {
		trigger_if ={
			limit = {
				is_subject = no
			}
			NOT = { has_government_type = gov_chartered_company }
		}
	}

	modifier = {
		country_legitimacy_headofstate_add = 100
	}

	ai_will_do = {
		exists = ruler
		ruler = {
			has_monarchist_ideology = yes
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

law_presidential_republic = {
	group = lawgroup_governance_principles

	icon = "gfx/interface/icons/law_icons/presidential_republic.dds"

	progressiveness = 100

	unlocking_technologies = {
		democracy
	}

	can_enact = {
		trigger_if ={
			limit = {
				is_subject = no
			}
			NOT = { has_government_type = gov_chartered_company }
		}
	}

	modifier = {
		country_legitimacy_headofstate_add = 100
	}

	ai_will_do = {
		exists = ruler
		OR = {
			ruler = {
				OR = {
					has_ideology = ideology:ideology_republican_leader
					has_ideology = ideology:ideology_radical
					has_ideology = ideology:ideology_positivist
					has_socialist_ideology = yes
				}
			}
			any_interest_group = {
				is_in_government = yes
				is_powerful = yes
				leader = {
					OR = {
						has_ideology = ideology:ideology_republican_leader
						has_ideology = ideology:ideology_radical
						has_ideology = ideology:ideology_positivist
						has_socialist_ideology = yes
					}
				}
			}
		}
	}

	ai_enact_weight_modifier = {
		value = 0
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

law_theocracy = {
	group = lawgroup_governance_principles

	icon = "gfx/interface/icons/law_icons/theocracy.dds"

	progressiveness = -25

	on_activate = {
		transition_off_of_monarchy_law_effects = yes # Reduced version of the effect that only removes Bakufu
	}

	modifier = {
		country_legitimacy_headofstate_add = 100
	}

	ai_will_do = {
		exists = ruler
		OR = {
			ruler = {
				has_ideology = ideology:ideology_theocrat
			}
			any_interest_group = {
				is_in_government = yes
				is_powerful = yes
				leader = {
					has_ideology = ideology:ideology_theocrat
				}
			}
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}
